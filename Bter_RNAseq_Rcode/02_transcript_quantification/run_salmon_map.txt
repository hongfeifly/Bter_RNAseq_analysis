#!/bin/sh

# run with fasta file
for name in ./reheader_dir/*.reheader.bam;
do

name_2="$(echo "$name" | cut -d '/' -f 3 | cut -d '.' -f 1)"
echo "$name_2"

salmon quant \
-l A \
-i ./index_dir/salmon_index/salmon_index \
-1 ${name} \
-2 ./data_dir/"$name_2".R2.fq.gz \
--validateMappings \
-o ./map_output_dir \
--gcBias

done
